CREATE OR REPLACE VIEW EDW.SALES_BRONZE.VW_SALES_BRONZE_VALID AS
SELECT 
  TRY_CAST(t.$1 AS VARCHAR)               AS SALES_ORDER_NUMBER,
  TRY_TO_NUMBER(t.$2, 38, 0)              AS SALES_ORDER_LINENUMBER,
  COALESCE(
    TRY_TO_DATE(TRIM(t.$3), 'YYYY-MM-DD'),
    TRY_TO_DATE(TRIM(t.$3), 'YYYY/MM/DD'),
    TRY_TO_DATE(TRIM(t.$3), 'DD/MM/YYYY'),
    TRY_TO_DATE(TRIM(t.$3), 'MM/DD/YYYY'),
    TRY_TO_DATE(TRIM(t.$3), 'DD-MON-YYYY'),
    TO_DATE(TRY_TO_TIMESTAMP_NTZ(TRIM(t.$3)))
  )                                       AS ORDER_DATE,
  TRY_CAST(t.$4 AS VARCHAR)               AS CUSTOMER_NAME,
  TRY_CAST(t.$5 AS VARCHAR)               AS EMAIL,
  TRY_CAST(t.$6 AS VARCHAR)               AS ITEM,
  TRY_TO_NUMBER(t.$7, 38, 0)              AS QUANTITY,
  TRY_TO_DECIMAL(t.$8, 18, 4)             AS UNITPRICE,
  TRY_TO_DECIMAL(t.$9, 18, 4)             AS TAX,

  d.MD5                                   AS FILE_CONTENT_KEY,
  METADATA$FILENAME                       AS FILE_NAME,
  CAST(d.LAST_MODIFIED AS TIMESTAMP_NTZ)  AS STG_MODIFIED_TS
FROM @EDW.SALES_BRONZE.MY_AWS_STAGE (
  FILE_FORMAT => EDW.SALES_BRONZE.CSV_FF_LOAD,
  PATTERN     => '^(.*)\\.csv$'
) t
LEFT JOIN DIRECTORY(@EDW.SALES_BRONZE.MY_AWS_STAGE) d
  ON d.RELATIVE_PATH = METADATA$FILENAME
WHERE
      TRY_CAST(t.$1 AS VARCHAR) IS NOT NULL
  AND TRY_TO_NUMBER(t.$2, 38, 0) IS NOT NULL
  AND COALESCE(
        TRY_TO_DATE(TRIM(t.$3), 'YYYY-MM-DD'),
        TRY_TO_DATE(TRIM(t.$3), 'YYYY/MM/DD'),
        TRY_TO_DATE(TRIM(t.$3), 'DD/MM/YYYY'),
        TRY_TO_DATE(TRIM(t.$3), 'MM/DD/YYYY'),
        TRY_TO_DATE(TRIM(t.$3), 'DD-MON-YYYY'),
        TO_DATE(TRY_TO_TIMESTAMP_NTZ(TRIM(t.$3)))
      ) IS NOT NULL
  AND TRY_TO_NUMBER(t.$7, 38, 0) > 0
  AND TRY_TO_DECIMAL(t.$8, 18, 4) >= 0;

  t.$1, t.$2, t.$3, t.$4, t.$5, t.$6, t.$7, t.$8, t.$9,
  METADATA$FILE_ROW_NUMBER,

  CASE
    WHEN TRY_TO_NUMBER(t.$2, 38, 0) IS NULL THEN FALSE

           TRY_TO_DATE(TRIM(t.$3), 'YYYY-MM-DD'),

           TRY_TO_DATE(TRIM(t.$3), 'DD/MM/YYYY'),

    WHEN TRY_TO_DECIMAL(t.$8, 18, 4) IS NULL THEN FALSE
    ELSE TRUE
  END AS IS_VALID_RECORD,
  CASE
    WHEN TRY_TO_NUMBER(t.$2, 38, 0) IS NULL THEN 'Invalid Line Number'
    WHEN COALESCE(
           TRY_TO_DATE(TRIM(t.$3), 'YYYY-MM-DD'),
           TRY_TO_DATE(TRIM(t.$3), 'YYYY/MM/DD'),
           TRY_TO_DATE(TRIM(t.$3), 'DD/MM/YYYY'),

           TO_DATE(TRY_TO_TIMESTAMP_NTZ(TRIM(t.$3)))

    WHEN TRY_TO_NUMBER(t.$7, 38, 0) IS NULL THEN 'Invalid Quantity'
    WHEN TRY_TO_DECIMAL(t.$8, 18, 4) IS NULL THEN 'Invalid Unit Price'

  END AS ERROR_REASON

---- SILVER TRANSFORMED TABLE 
CREATE OR REPLACE TABLE EDW.SALES_SILVER.PROCESSED_SALES_DATA (
  SALES_ORDER_NUMBER      VARCHAR,
  SALES_ORDER_LINENUMBER  NUMBER(38,0),
  ORDER_DATE              DATE,
  CUSTOMER_NAME           VARCHAR,
  EMAIL                   VARCHAR,
  ITEM                    VARCHAR,
  QUANTITY                NUMBER(38,0),
  UNITPRICE               NUMBER(18,4),
  TAX                     NUMBER(18,4),
  EMAIL_NORM              VARCHAR,
  CUSTOMER_NAME_CLEAN     VARCHAR,
  ITEM_CLEAN              VARCHAR,
  ORDER_DATE_ID           NUMBER(8,0),
  ORDER_YYYYMM            NUMBER(6,0),
  NET_SALES               NUMBER(18,4),
  GROSS_SALES             NUMBER(18,4),
  FILE_ROW_NUMBER         NUMBER(38,0),
  FILE_CONTENT_KEY        VARCHAR,
  FILE_NAME               VARCHAR,
  CREATED_TS              TIMESTAMP_NTZ,
  MODIFIED_TS             TIMESTAMP_NTZ,
  STG_MODIFIED_TS         TIMESTAMP_NTZ
) DATA_RETENTION_TIME_IN_DAYS = 90;

MERGE INTO EDW.SALES_SILVER.PROCESSED_SALES_DATA AS tgt
USING (
  SELECT *
  FROM (
    SELECT
      v.SALES_ORDER_NUMBER,
      v.SALES_ORDER_LINENUMBER,
      v.ORDER_DATE,
      v.CUSTOMER_NAME,
      v.EMAIL,
      v.ITEM,
      v.QUANTITY::NUMBER(38,0)                       AS QUANTITY,
      ROUND(v.UNITPRICE, 4)                          AS UNITPRICE,
      COALESCE(v.TAX, 0)                             AS TAX,
      LOWER(TRIM(v.EMAIL))                           AS EMAIL_NORM,
      INITCAP(REGEXP_REPLACE(TRIM(v.CUSTOMER_NAME), '\\s+', ' ')) AS CUSTOMER_NAME_CLEAN,
      UPPER(REGEXP_REPLACE(TRIM(v.ITEM), '\\s+', ' '))            AS ITEM_CLEAN,
      TO_NUMBER(TO_CHAR(v.ORDER_DATE, 'YYYYMMDD'))   AS ORDER_DATE_ID,
      TO_NUMBER(TO_CHAR(v.ORDER_DATE, 'YYYYMM'))     AS ORDER_YYYYMM,
      ROUND(v.QUANTITY * v.UNITPRICE, 4)             AS NET_SALES,
      ROUND(v.QUANTITY * v.UNITPRICE + NVL(v.TAX,0), 4) AS GROSS_SALES,
      v.FILE_ROW_NUMBER,
      v.FILE_CONTENT_KEY,
      v.FILE_NAME,
      CURRENT_TIMESTAMP()                            AS CREATED_TS,
      CURRENT_TIMESTAMP()                            AS MODIFIED_TS,
      v.STG_MODIFIED_TS,
      ROW_NUMBER() OVER (
        PARTITION BY v.SALES_ORDER_NUMBER, v.SALES_ORDER_LINENUMBER, v.ORDER_DATE, v.EMAIL, v.ITEM
        ORDER BY v.STG_MODIFIED_TS DESC, v.FILE_ROW_NUMBER DESC
      ) AS RN
    FROM EDW.SALES_BRONZE.VW_SALES_BRONZE_VALID v
  )
  WHERE RN = 1
) AS src
ON  tgt.SALES_ORDER_NUMBER     = src.SALES_ORDER_NUMBER
AND tgt.SALES_ORDER_LINENUMBER = src.SALES_ORDER_LINENUMBER
AND tgt.ORDER_DATE             = src.ORDER_DATE
AND tgt.EMAIL                  = src.EMAIL
AND tgt.ITEM                   = src.ITEM
WHEN MATCHED THEN UPDATE SET
  CUSTOMER_NAME       = src.CUSTOMER_NAME,
  QUANTITY            = src.QUANTITY,
  UNITPRICE           = src.UNITPRICE,
  TAX                 = src.TAX,
  EMAIL_NORM          = src.EMAIL_NORM,
  CUSTOMER_NAME_CLEAN = src.CUSTOMER_NAME_CLEAN,
  ITEM_CLEAN          = src.ITEM_CLEAN,
  ORDER_DATE_ID       = src.ORDER_DATE_ID,
  ORDER_YYYYMM        = src.ORDER_YYYYMM,
  NET_SALES           = src.NET_SALES,
  GROSS_SALES         = src.GROSS_SALES,
  FILE_ROW_NUMBER     = src.FILE_ROW_NUMBER,
  FILE_CONTENT_KEY    = src.FILE_CONTENT_KEY,
  FILE_NAME           = src.FILE_NAME,
  MODIFIED_TS         = src.MODIFIED_TS,
  STG_MODIFIED_TS     = src.STG_MODIFIED_TS
WHEN NOT MATCHED THEN INSERT (
  SALES_ORDER_NUMBER, SALES_ORDER_LINENUMBER, ORDER_DATE, CUSTOMER_NAME, EMAIL, ITEM,
  QUANTITY, UNITPRICE, TAX,
  EMAIL_NORM, CUSTOMER_NAME_CLEAN, ITEM_CLEAN, ORDER_DATE_ID, ORDER_YYYYMM,
  NET_SALES, GROSS_SALES,
  FILE_ROW_NUMBER, FILE_CONTENT_KEY, FILE_NAME, CREATED_TS, MODIFIED_TS, STG_MODIFIED_TS
) VALUES (
  src.SALES_ORDER_NUMBER, src.SALES_ORDER_LINENUMBER, src.ORDER_DATE, src.CUSTOMER_NAME, src.EMAIL, src.ITEM,
  src.QUANTITY, src.UNITPRICE, src.TAX,
  src.EMAIL_NORM, src.CUSTOMER_NAME_CLEAN, src.ITEM_CLEAN, src.ORDER_DATE_ID, src.ORDER_YYYYMM,
  src.NET_SALES, src.GROSS_SALES,
  src.FILE_ROW_NUMBER, src.FILE_CONTENT_KEY, src.FILE_NAME, src.CREATED_TS, src.MODIFIED_TS, src.STG_MODIFIED_TS
);

-- Gold Layer ( Load the Data to Bronze Layer )

USE SCHEMA EDW.SALES_GOLD;

CREATE OR REPLACE TABLE EDW.SALES_GOLD.DIM_CUSTOMER (
  CUSTOMER_ID  NUMBER(38,0) AUTOINCREMENT START 1 INCREMENT 1 PRIMARY KEY,
  CUSTOMER_NAME VARCHAR(100),
  EMAIL         VARCHAR(100),
  EMAIL_NORM    VARCHAR(100)
);

CREATE OR REPLACE TABLE EDW.SALES_GOLD.DIM_PRODUCT (
  ITEM_ID    NUMBER(38,0) AUTOINCREMENT START 1 INCREMENT 1 PRIMARY KEY,
  ITEM_NAME  VARCHAR(200),
  ITEM_CLEAN VARCHAR(200)
);

CREATE OR REPLACE TABLE EDW.SALES_GOLD.DIM_DATE (
  ORDER_DATE_ID NUMBER(38,0) PRIMARY KEY,
  ORDER_DATE    DATE,
  YEAR          NUMBER(4,0),
  MONTH         NUMBER(2,0),
  DAY           NUMBER(2,0),
  YYYYMM        VARCHAR(6)
);

CREATE OR REPLACE TABLE EDW.SALES_GOLD.FACT_SALES (
  SALES_ORDER_NUMBER VARCHAR,
  CUSTOMER_ID        NUMBER(38,0),
  ITEM_ID            NUMBER(38,0),
  ORDER_DATE_ID      NUMBER(38,0),
  QUANTITY           NUMBER(38,0),
  UNIT_PRICE         NUMBER(18,4),
  TAX                NUMBER(18,4),
  NET_SALES          NUMBER(18,4),
  GROSS_SALES        NUMBER(18,4),
  FILE_ROW_NUMBER    NUMBER(38,0),
  FILE_CONTENT_KEY   VARCHAR,
  FILE_NAME          VARCHAR,
  FOREIGN KEY (CUSTOMER_ID)   REFERENCES EDW.SALES_GOLD.DIM_CUSTOMER(CUSTOMER_ID),
  FOREIGN KEY (ITEM_ID)       REFERENCES EDW.SALES_GOLD.DIM_PRODUCT(ITEM_ID),
  FOREIGN KEY (ORDER_DATE_ID) REFERENCES EDW.SALES_GOLD.DIM_DATE(ORDER_DATE_ID)
);

MERGE INTO EDW.SALES_GOLD.DIM_CUSTOMER d
USING (
  SELECT DISTINCT CUSTOMER_NAME_CLEAN AS CUSTOMER_NAME, EMAIL, EMAIL_NORM
  FROM EDW.SALES_SILVER.PROCESSED_SALES_DATA
) s
ON d.EMAIL = s.EMAIL AND d.EMAIL_NORM = s.EMAIL_NORM
WHEN MATCHED THEN UPDATE SET d.CUSTOMER_NAME = s.CUSTOMER_NAME
WHEN NOT MATCHED THEN INSERT (CUSTOMER_NAME, EMAIL, EMAIL_NORM)
VALUES (s.CUSTOMER_NAME, s.EMAIL, s.EMAIL_NORM);

MERGE INTO EDW.SALES_GOLD.DIM_PRODUCT p
USING (
  SELECT DISTINCT ITEM AS ITEM_NAME, ITEM_CLEAN
  FROM EDW.SALES_SILVER.PROCESSED_SALES_DATA
) s
ON p.ITEM_NAME = s.ITEM_NAME AND p.ITEM_CLEAN = s.ITEM_CLEAN
WHEN NOT MATCHED THEN INSERT (ITEM_NAME, ITEM_CLEAN)
VALUES (s.ITEM_NAME, s.ITEM_CLEAN);

MERGE INTO EDW.SALES_GOLD.DIM_DATE dd
USING (
  SELECT DISTINCT
    ORDER_DATE,
    TO_NUMBER(TO_CHAR(ORDER_DATE, 'YYYYMMDD')) AS ORDER_DATE_ID,
    YEAR(ORDER_DATE)  AS YEAR,
    MONTH(ORDER_DATE) AS MONTH,
    DAY(ORDER_DATE)   AS DAY,
    TO_VARCHAR(ORDER_DATE, 'YYYYMM') AS YYYYMM
  FROM EDW.SALES_SILVER.PROCESSED_SALES_DATA
) s
ON dd.ORDER_DATE_ID = s.ORDER_DATE_ID
WHEN NOT MATCHED THEN INSERT (ORDER_DATE_ID, ORDER_DATE, YEAR, MONTH, DAY, YYYYMM)
VALUES (s.ORDER_DATE_ID, s.ORDER_DATE, s.YEAR, s.MONTH, s.DAY, s.YYYYMM);

INSERT INTO EDW.SALES_GOLD.FACT_SALES (
  SALES_ORDER_NUMBER, CUSTOMER_ID, ITEM_ID, ORDER_DATE_ID,
  QUANTITY, UNIT_PRICE, TAX, NET_SALES, GROSS_SALES,
  FILE_ROW_NUMBER, FILE_CONTENT_KEY, FILE_NAME
)
SELECT
  s.SALES_ORDER_NUMBER,
  dc.CUSTOMER_ID,
  dp.ITEM_ID,
  s.ORDER_DATE_ID,
  s.QUANTITY,
  s.UNITPRICE,
  s.TAX,
  s.NET_SALES,
  s.GROSS_SALES,
  s.FILE_ROW_NUMBER,
  s.FILE_CONTENT_KEY,
  s.FILE_NAME
FROM EDW.SALES_SILVER.PROCESSED_SALES_DATA s
JOIN EDW.SALES_GOLD.DIM_CUSTOMER dc ON dc.EMAIL_NORM = s.EMAIL_NORM
JOIN EDW.SALES_GOLD.DIM_PRODUCT  dp ON dp.ITEM_CLEAN = s.ITEM_CLEAN
JOIN EDW.SALES_GOLD.DIM_DATE     dd ON dd.ORDER_DATE_ID = s.ORDER_DATE_ID;

USE DATABASE EDW;

WITH
bronze_valid AS ( SELECT COUNT(*) AS cnt FROM EDW.SALES_BRONZE.VW_SALES_BRONZE_VALID ),
bronze_error AS ( SELECT COUNT(*) AS cnt FROM EDW.SALES_BRONZE.VW_SALES_BRONZE_ERROR ),
silver_audit AS ( SELECT COUNT(*) AS cnt FROM EDW.SALES_SILVER.SILVER_RAW_AUDIT ),
silver_ok    AS ( SELECT COUNT(*) AS cnt FROM EDW.SALES_SILVER.PROCESSED_SALES_DATA ),
silver_err   AS ( SELECT COUNT(*) AS cnt FROM EDW.SALES_SILVER.PROCESSED_SALES_DATA_ERROR ),
gold_dim_c   AS ( SELECT COUNT(*) AS cnt FROM EDW.SALES_GOLD.DIM_CUSTOMER ),
gold_dim_p   AS ( SELECT COUNT(*) AS cnt FROM EDW.SALES_GOLD.DIM_PRODUCT ),
gold_dim_d   AS ( SELECT COUNT(*) AS cnt FROM EDW.SALES_GOLD.DIM_DATE ),
gold_fact    AS ( SELECT COUNT(*) AS cnt FROM EDW.SALES_GOLD.FACT_SALES )
SELECT 'BRONZE' AS LAYER, 'valid_rows' AS CHECK_NAME, cnt AS VALUE FROM bronze_valid
UNION ALL
SELECT 'BRONZE', 'error_rows', cnt FROM bronze_error
UNION ALL
SELECT 'SILVER', 'audit_rows', cnt FROM silver_audit
UNION ALL
